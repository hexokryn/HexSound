cmake_minimum_required(VERSION 3.21)
project(HexSound VERSION 0.13.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 WebEngineQuick Network Test)
if(MSVC)
 add_compile_options(/utf-8)
endif()
qt_add_executable(HexSound native/main.cpp native/SmokeHarness.h native/Controller.cpp native/Controller.h native/DiscordPresence.cpp native/DiscordPresence.h native/MusicData.cpp native/MusicData.h native/Services.cpp native/Services.h)
target_link_libraries(HexSound PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick Qt6::Network)
if(WIN32)
 target_link_libraries(HexSound PRIVATE crypt32)
 set_target_properties(HexSound PROPERTIES WIN32_EXECUTABLE TRUE)
 configure_file(native/windows.rc.in windows.rc @ONLY)
 target_sources(HexSound PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
endif()
target_compile_definitions(HexSound PRIVATE HEXSOUND_VERSION="${PROJECT_VERSION}")
qt_add_resources(HexSound app_resources PREFIX "/" FILES native/qml/Main.qml native/qml/HSButton.qml native/qml/Setting.qml native/qml/Atmosphere.qml native/qml/HSSelect.qml native/qml/HSField.qml native/qml/HSSlider.qml native/qml/HSSwitch.qml native/qml/HSIcon.qml native/assets/mark-light.svg native/assets/palettes.json native/assets/soundcloud.css native/assets/collector.js native/assets/remote-player.js native/assets/web-player.js native/assets/mark.svg native/assets/icon.png)
include(CTest)
if(BUILD_TESTING)
 qt_add_executable(update_probe native/tests/UpdateProbe.cpp)
 target_link_libraries(update_probe PRIVATE Qt6::Core)
 qt_add_executable(rpc_tests native/tests/tst_rpc.cpp native/DiscordPresence.cpp native/DiscordPresence.h)
 target_link_libraries(rpc_tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
 add_test(NAME discord_rpc COMMAND rpc_tests)
 qt_add_executable(native_tests native/tests/tst_music.cpp native/MusicData.cpp native/MusicData.h)
 target_link_libraries(native_tests PRIVATE Qt6::Core Qt6::Test)
 add_test(NAME native_music COMMAND native_tests)
 qt_add_executable(controller_tests native/tests/tst_controller.cpp native/Controller.cpp native/Controller.h native/DiscordPresence.cpp native/DiscordPresence.h native/Services.cpp native/Services.h native/MusicData.cpp native/MusicData.h)
 target_link_libraries(controller_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Test)
 target_compile_definitions(controller_tests PRIVATE HEXSOUND_VERSION="${PROJECT_VERSION}")
 if(WIN32)
  target_link_libraries(controller_tests PRIVATE crypt32)
 endif()
 add_test(NAME native_controller COMMAND controller_tests)
endif()
include(GNUInstallDirs)
if(WIN32)
 set(app_runtime_destination .)
else()
 set(app_runtime_destination ${CMAKE_INSTALL_BINDIR})
 set_target_properties(HexSound PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS HexSound RUNTIME DESTINATION ${app_runtime_destination} BUNDLE DESTINATION .)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY native/licenses DESTINATION .)
qt_generate_deploy_qml_app_script(TARGET HexSound OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR NO_TRANSLATIONS)
install(SCRIPT ${deploy_script})
if(WIN32)
 set(CPACK_GENERATOR "ZIP;NSIS")
 set(CPACK_PACKAGE_FILE_NAME "HexSound-${PROJECT_VERSION}-windows-x64")
 set(CPACK_NSIS_DISPLAY_NAME "HexSound")
 set(CPACK_NSIS_PACKAGE_NAME "HexSound")
 set(CPACK_PACKAGE_EXECUTABLES "HexSound" "HexSound")
 set(CPACK_NSIS_EXECUTABLES_DIRECTORY ".")
else()
 install(FILES native/hexsound.desktop DESTINATION share/applications)
 install(FILES native/assets/icon.png DESTINATION share/icons/hicolor/256x256/apps RENAME hexsound.png)
 set(CPACK_GENERATOR "TGZ;DEB")
 set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
 set(CPACK_DEBIAN_PACKAGE_FILE_NAME "HexSound-${PROJECT_VERSION}-linux-amd64.deb")
 set(CPACK_DEBIAN_PACKAGE_MAINTAINER "zexokky")
endif()
set(CPACK_PACKAGE_NAME "HexSound")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/release")
set(CPACK_PACKAGE_VENDOR "zexokky")
set(CPACK_PACKAGE_CONTACT "https://t.me/zexokkydev")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY "HexSound")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
include(CPack)

qt_add_executable(HexSoundUpdate native/UpdateHelper.cpp)
if(WIN32)
 set_target_properties(HexSoundUpdate PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
target_link_libraries(HexSoundUpdate PRIVATE Qt6::Core)
install(TARGETS HexSoundUpdate RUNTIME DESTINATION ${app_runtime_destination})
